3.1 The Security Jail (core/security.py)
Prevents CWE-22 Path Traversal. No stage or provider can touch a file outside the job directory.

python

import os
from pathlib import Path
from magnific.core.errors import SecurityViolationError

class WorkspaceManager:
    """Strict filesystem jail. All paths MUST be resolved through this."""
    
    def __init__(self, base_dir: Path):
        # Resolve to absolute path to prevent relative traversal tricks
        self._root = base_dir.resolve()
        self._root.mkdir(parents=True, exist_ok=True)

    @property
    def root(self) -> Path:
        return self._root

    def resolve(self, *parts: str, mkdir: bool = False) -> Path:
        """Constructs a path safely inside the jail."""
        # Join and resolve (eliminates ../)
        target = (self._root / Path(*parts)).resolve()
        
        # Verify the resolved path is STILL inside the root
        try:
            target.relative_to(self._root)
        except ValueError as e:
            raise SecurityViolationError(
                f"Path traversal blocked: '{parts}' escapes job workspace."
            ) from e
            
        if mkdir:
            target.mkdir(parents=True, exist_ok=True)
            
        return target
3.2 Ghost Job Defense (core/tracking.py)
If the server crashes during Veo polling, we don't lose the operation IDs.

python

import json
import time
from pathlib import Path
from magnific.core.security import WorkspaceManager

class OperationTracker:
    """Append-only state store for cloud operations."""
    
    def __init__(self, workspace: WorkspaceManager):
        self._file = workspace.resolve("pending_operations.jsonl")

    def record_submission(self, scene_id: str, operation_id: str) -> None:
        """Writes immediately. Append-only is extremely crash-resilient."""
        payload = {"scene_id": scene_id, "op_id": operation_id, "ts": time.time()}
        with open(self._file, "a", encoding="utf-8") as f:
            f.write(json.dumps(payload) + "\n")

    def get_pending_operations(self) -> dict[str, str]:
        """On resume, find operations that never finished."""
        if not self._file.exists():
            return {}
        
        pending = {}
        for line in self._file.read_text(encoding="utf-8").strip().splitlines():
            if line:
                data = json.loads(line)
                pending[data["scene_id"]] = data["op_id"]
        return pending

    def mark_complete(self, scene_id: str) -> None:
        """Removes a scene from tracking once MP4 is safely local."""
        pending = self.get_pending_operations()
        pending.pop(scene_id, None)
        
        # Overwrite file with remaining items
        self._file.write_text(
            "\n".join(json.dumps({"scene_id": s, "op_id": o}) for s, o in pending.items()),
            encoding="utf-8"
        )
3.3 The Error Classifier (core/retry.py)
The exact matrix ensuring Safety Filters are NEVER retried.

python

import enum
from google.api_core import exceptions as gcp_errors

class ErrorAction(enum.Enum):
    RETRY = "retry"
    FAIL_SCENE = "fail_scene"   # Permanent error, but pipeline continues
    ABORT_JOB = "abort_job"     # Unrecoverable state error

class ErrorClassifier:
    """Maps Google API exceptions to specific architectural actions."""
    
    @staticmethod
    def classify(error: Exception) -> ErrorAction:
        if isinstance(error, gcp_errors.ResourceExhausted): # 429
            return ErrorAction.RETRY
        if isinstance(error, (gcp_errors.ServiceUnavailable, gcp_errors.DeadlineExceeded)): # 503/504
            return ErrorAction.RETRY
            
        # PERMANENT FAILURES - Do not waste API quota retrying these
        if isinstance(error, gcp_errors.InvalidArgument): # 400
            return ErrorAction.FAIL_SCENE
        if isinstance(error, gcp_errors.PermissionDenied): # 403 (Safety blocks often manifest here)
            return ErrorAction.FAIL_SCENE
            
        # Default: Fail closed. Unknown errors are not retried.
        return ErrorAction.FAIL_SCENE
3.4 Pre-flight Gate (core/preflight.py)
Stops the pipeline before it wastes money or fills the disk.

python

import shutil
from pathlib import Path
from PIL import Image
from magnific.core.errors import PreFlightError
from magnific.core.security import WorkspaceManager

class PreFlightChecker:
    def __init__(self, workspace: WorkspaceManager, min_disk_gb: float, max_img_px: int):
        self.workspace = workspace
        self.min_disk_bytes = min_disk_gb * 1024**3
        self.max_px = max_img_px

    def validate_environment(self) -> None:
        """Run all pre-flight checks."""
        self._check_disk()
        
    def process_reference_images(self, paths: list[Path]) -> list[Path]:
        """Optimize images to prevent token bloat and memory spikes."""
        out_paths = []
        refs_dir = self.workspace.resolve("refs", mkdir=True)
        
        for i, p in enumerate(paths):
            if not p.exists(): raise PreFlightError(f"Reference image missing: {p}")
            
            out_path = refs_dir / f"char_{i+1}_optimized.jpg"
            
            with Image.open(p) as img:
                # Scale down if too large
                if max(img.size) > self.max_px:
                    ratio = self.max_px / max(img.size)
                    img = img.resize((int(img.width * ratio), int(img.height * ratio)))
                
                # Strip EXIF, compress to JPEG
                img.convert("RGB").save(out_path, "JPEG", quality=85, optimize=True)
                
            out_paths.append(out_path)
            
        return out_paths

    def _check_disk(self) -> None:
        usage = shutil.disk_usage(self.workspace.root)
        if usage.free < self.min_disk_bytes:
            raise PreFlightError(
                f"Insufficient disk space. Need {self.min_disk_bytes/1024**3:.1f}GB, "
                f"have {usage.free/1024**3:.1f}GB."
            )
3.5 The Hybrid Base Stage (stages/base.py)
How we handle Kimi's reality check: Sync SDKs wrapped in Async threads.

python

import asyncio
import os
from pathlib import Path
from abc import ABC, abstractmethod
from magnific.core.security import WorkspaceManager
from magnific.core.manifest import BaseManifest

class BaseStage(ABC):
    def __init__(self, workspace: WorkspaceManager):
        self.workspace = workspace

    @abstractmethod
    async def execute(self, input_manifest: BaseManifest | None) -> BaseManifest:
        pass

    def _atomic_write_manifest(self, manifest: BaseManifest, filename: str) -> None:
        """Prevents corrupt manifests if power is pulled mid-write."""
        target = self.workspace.resolve(filename)
        temp = target.with_suffix(".json.tmp")
        
        # Write to temp
        temp.write_text(manifest.to_json(), encoding="utf-8")
        
        # Atomic rename (virtually instantaneous on POSIX/NTFS)
        os.replace(temp, target)

    @staticmethod
    async def _run_sync_in_thread(sync_func, *args, **kwargs):
        """
        The Hybrid Concurrency Pattern.
        Google's genai/imagen SDKs are blocking. We run them in a 
        ThreadPoolExecutor so they don't freeze the asyncio event loop.
        """
        loop = asyncio.get_running_loop()
        return await loop.run_in_executor(None, lambda: sync_func(*args, **kwargs))
4. The Execution Blueprint (Step-by-Step)
Here is exactly what happens when you run:
python -m magnific run --config workflow.yaml

Step 0: Boot & Security Gate
CLI loads workflow.yaml, merges with default.yaml.
Orchestrator creates Job (UUID).
Orchestrator instantiates WorkspaceManager (Jail).
Orchestrator runs PreFlightChecker. If disk is full or images are missing -> HALT.
Reference images are optimized and copied into the jailed /refs/ directory.
Step 1: Story Generation (Sync in Thread)
StoryStage loads optimized images as base64 (memory safe, they are small now).
Calls Gemini via await self._run_sync_in_thread(gemini.generate, ...).
Parses JSON. If schema invalid -> ErrorAction.RETRY (once).
Writes story_manifest.json atomically.
Step 2: Preview Generation (Cached Concurrency)
PreviewStage loads story_manifest.json.
Creates asyncio.Semaphore(max_concurrent=3).
Loops through scenes. Inside the loop:
a. Acquire semaphore.
b. Fetch image bytes via GoogleImagenProvider.get_cached_bytes() (prevents 10x memory duplication).
c. Call Imagen via await self._run_sync_in_thread(imagen.generate, ...).
If ErrorClassifier returns FAIL_SCENE (Safety block), catch it, mark scene failed in manifest, release semaphore, continue.
Writes preview_manifest.json atomically.
Provider calls .clear_cache() to free memory before Stage 3.
Step 3: Video Animation (Async Polling + Crash Recovery)
VideoStage loads preview_manifest.json (filters to status: success).
Initializes OperationTracker.
Checks pending_operations.jsonl for crashed jobs.
Submission Phase: For each scene, calls Veo via thread. The microsecond we get an operation_id back, we call tracker.record_submission().
Polling Phase: Native asyncio.gather() polls all operation IDs concurrently.
When a video succeeds, we download MP4 to jailed /videos/ dir, then call tracker.mark_complete().
Writes video_manifest.json atomically.
The Crash Scenario Test
Power cord pulled at Step 3.5 (halfway through polling).

What happens on restart: User runs --from-stage video.
System loads: preview_manifest.json (knows 6 scenes need video).
System loads: pending_operations.jsonl (finds 3 operation IDs still logged).
Action: System submits 3 new videos, and resumes polling the 3 existing ones. Zero duplicate costs. Zero ghost jobs.

Final Architectural Sign-Off
This design is no longer just a "pipeline." It is a resilient state machine.

By implementing the Path Jail, we made it secure against config injection. By implementing Atomic Writes, we made it immune to corrupt state. By implementing Operation Tracking, we eliminated the possibility of cloud resource leaks. And by implementing the Hybrid Threading Model, we acknowledged the reality of Google's Python SDKs without sacrificing the concurrency required for performant video polling.
